
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include "sort.h"

int main() {
    int sizes[] = {10, 1000, 10000, 100000};
    int num_size = 4;

    printf("ArrSize\tBubble\tMerge\tInsertion\n");

    srand(time(NULL));

    for (int a = 0; a < num_size; a++) {
        int n = sizes[a];

        int *source = (int*)malloc(n * sizeof(int));
        int *arr = (int*)malloc(n * sizeof(int));
        int *temp = (int*)malloc(n * sizeof(int));

        if (source == NULL || arr == NULL || temp == NULL) {
            printf("Array_error\n");
            return 1;
        }

        for (int i = 0; i < n; i++) {
            source[i] = rand();
        }

        double t_b, t_m, t_i;
        clock_t start, end;

        // пузырьковая сортировка
        for (int i = 0; i < n; i++) {
            arr[i] = source[i];
        }

        start = clock();
        bubble_sort(arr, n);
        end = clock();
        t_b = (double)(end - start) / CLOCKS_PER_SEC;

        // сортировка слиянием
        for (int i = 0; i < n; i++) {
            arr[i] = source[i];
        }

        start = clock();
        merge_sort(arr, temp, 0, n);
        end = clock();
        t_m = (double)(end - start) / CLOCKS_PER_SEC;

        // сортировка вставками
        for (int i = 0; i < n; i++) {
            arr[i] = source[i];
        }

        start = clock();
        insertion_sort(arr, n);
        end = clock();
        t_i = (double)(end - start) / CLOCKS_PER_SEC;

        printf("%d\t%f\t%f\t%f\n", n, t_b, t_m, t_i);

        free(source);
        free(arr);
        free(temp);
    }

    return 0;
}
